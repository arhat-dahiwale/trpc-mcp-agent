# DATA.md

The published packages have **no database**. This document covers (1) the library's in-memory data types and (2) the persistent schema for the example backend and playground (PostgreSQL via Drizzle, plus Redis).

---

## 1. Library types (`@trpc-mcp/core`)

```ts
/** Attached to procedures via .meta({ mcp: {...} }) */
export interface McpMeta {
  mcp?: {
    expose?: boolean;            // default false (opt-in)
    name?: string;               // override derived tool name
    title?: string;
    description?: string;        // falls back to procedure description
    tags?: string[];
    annotations?: {
      readOnlyHint?: boolean;
      destructiveHint?: boolean;
      idempotentHint?: boolean;
      openWorldHint?: boolean;
    };
    longRunning?: boolean;       // enables progress contract
    timeoutMs?: number;
  };
}

export type ProcedureType = "query" | "mutation";

export interface ToolDefinition {
  name: string;                  // e.g. "notes_search"
  path: string;                  // e.g. "notes.search"
  type: ProcedureType;
  title?: string;
  description: string;
  inputSchema: JsonSchema;       // JSON Schema 2020-12 object schema
  outputSchema?: JsonSchema;
  annotations: Required<NonNullable<McpMeta["mcp"]>>["annotations"];
  tags: string[];
  longRunning: boolean;
  timeoutMs?: number;
}

export interface ToolCallResult {
  ok: boolean;
  data?: unknown;
  error?: { code: string; message: string };   // sanitized
  durationMs: number;
}

export type ToolPolicy<Ctx> = (
  ctx: Ctx,
  tool: Pick<ToolDefinition, "name" | "path" | "type" | "tags">
) => boolean | Promise<boolean>;

export interface RegistryOptions<Router, Ctx> {
  router: Router;
  createContext: (info: TransportInfo) => Ctx | Promise<Ctx>;
  policy?: ToolPolicy<Ctx>;
  include?: string[];            // glob patterns on path
  exclude?: string[];
  nameStrategy?: (path: string) => string;
  errorFormatter?: (err: unknown) => { code: string; message: string };
}

export type TransportInfo =
  | { transport: "stdio"; env: NodeJS.ProcessEnv }
  | { transport: "http"; request: Request }
  | { transport: "ai-sdk"; options?: unknown };
```

### Naming rules
- Path `a.b.c` becomes `a_b_c`.
- Allowed characters: `[a-zA-Z0-9_-]`, max 64 chars.
- Collision or invalid name: throw at registry creation with both paths in the message.

## 2. Example backend: PostgreSQL (Drizzle)

Better Auth manages the identity tables. Table and column names below are the intended shape. Confirm against the Better Auth schema generator output when implementing.

### Auth (generated by Better Auth + organization plugin)
`user`, `session`, `account`, `verification`, `organization`, `member`, `invitation`

`member.role` values: `owner | admin | member | viewer`

### Application tables

```sql
-- Example domain: notes + tasks, used to demo tools
CREATE TABLE notes (
  id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  organization_id text NOT NULL REFERENCES organization(id) ON DELETE CASCADE,
  author_id     text NOT NULL REFERENCES "user"(id),
  title         text NOT NULL,
  body          text NOT NULL DEFAULT '',
  tags          text[] NOT NULL DEFAULT '{}',
  created_at    timestamptz NOT NULL DEFAULT now(),
  updated_at    timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX notes_org_idx ON notes (organization_id, created_at DESC);

CREATE TABLE tasks (
  id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  organization_id text NOT NULL REFERENCES organization(id) ON DELETE CASCADE,
  assignee_id   text REFERENCES "user"(id),
  title         text NOT NULL,
  status        text NOT NULL CHECK (status IN ('todo','doing','done')) DEFAULT 'todo',
  due_at        timestamptz,
  created_at    timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE report_jobs (            -- long-running tool demo
  id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  organization_id text NOT NULL REFERENCES organization(id) ON DELETE CASCADE,
  requested_by  text NOT NULL REFERENCES "user"(id),
  bullmq_job_id text,
  status        text NOT NULL CHECK (status IN ('queued','running','completed','failed','cancelled')) DEFAULT 'queued',
  progress      integer NOT NULL DEFAULT 0 CHECK (progress BETWEEN 0 AND 100),
  result_url    text,
  error         text,
  created_at    timestamptz NOT NULL DEFAULT now(),
  finished_at   timestamptz
);
```

### Playground tables

```sql
CREATE TABLE tool_calls (             -- audit log of every tool invocation
  id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  organization_id text REFERENCES organization(id) ON DELETE SET NULL,
  user_id       text REFERENCES "user"(id) ON DELETE SET NULL,
  source        text NOT NULL CHECK (source IN ('playground','chat','mcp-http','mcp-stdio')),
  tool_name     text NOT NULL,
  input         jsonb NOT NULL,
  output        jsonb,
  status        text NOT NULL CHECK (status IN ('ok','error','denied')),
  error_code    text,
  duration_ms   integer NOT NULL,
  trace_id      text,
  created_at    timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX tool_calls_org_time_idx ON tool_calls (organization_id, created_at DESC);
CREATE INDEX tool_calls_tool_idx ON tool_calls (tool_name, created_at DESC);

CREATE TABLE chat_threads (
  id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  organization_id text NOT NULL REFERENCES organization(id) ON DELETE CASCADE,
  user_id       text NOT NULL REFERENCES "user"(id) ON DELETE CASCADE,
  title         text,
  created_at    timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE chat_messages (
  id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  thread_id     uuid NOT NULL REFERENCES chat_threads(id) ON DELETE CASCADE,
  role          text NOT NULL CHECK (role IN ('user','assistant','tool','system')),
  parts         jsonb NOT NULL,        -- AI SDK message parts
  created_at    timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX chat_messages_thread_idx ON chat_messages (thread_id, created_at);
```

## 3. Redis

| Key / structure | Purpose | TTL |
|---|---|---|
| `bull:reports:*` | BullMQ queue internals | managed by BullMQ |
| `progress:{jobId}` | Latest progress snapshot (JSON) | 1 hour |
| `rate:{userId}:{tool}` | Sliding-window rate limit counter | 60 s |
| Socket.IO adapter channels | Fan-out of progress events | n/a |
